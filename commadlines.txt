#include "get_next_line.h"
#include <fcntl.h>
#include <stdio.h>

int	main(int argc, char **argv)
{
	int		fd;
	char	*line;
	int		i;

	fd = open(argv[1], O_RDONLY);
	i = 1;
	while ((line = get_next_line(fd)) != NULL)
	{
		printf("CALL %d:\n", i);
		printf("%s", line);
		printf("----------------\n");
		free(line);
		i++;
	}
	close(fd);
	return (0);
}
